# Sample Notification - Cards

```yaml
openapi: 3.0.1
info:
  title: ''
  description: ''
  version: 1.0.0
paths:
  /merchants-url/receive-notification:
    post:
      summary: Sample Notification - Cards
      deprecated: false
      description: >-
        After setting up your Notification URL, we will send the notification
        for certain events such as when the Credit Card has been charged, Credit
        Card failed, and etc.


        Here is the sample notification that DOKU will send to merchant's
        server.
      tags:
        - Notification
      parameters:
        - name: Client-Id
          in: header
          description: Client ID you retrieved from DOKU Back Office
          required: true
          example: MCH-0001-10791114622547
          schema:
            type: string
        - name: Request-Id
          in: header
          description: >-
            Unique random string (max 128 characters) that generated by DOKU to
            make sure no duplicate notification
          required: true
          example: 479b663f-5c9d-400d-8e80-3e548a8f7639
          schema:
            type: string
        - name: Request-Timestamp
          in: header
          description: Time Stamp request on UTC time in ISO8601 format
          required: true
          example: '2020-08-11T08:45:42Z'
          schema:
            type: string
        - name: Signature
          in: header
          description: >-
            HMACSHA256 from signature components, to generate signature please
            refer to
            [this](https://doku-developers.apidog.io/signature-component-from-request-header-1387601m0)
            page.
          required: true
          example: HMACSHA256=vl9DBTX5KhEiXmnpOD0TSm8PYQknuHPdyHSTSc3W6Ps=
          schema:
            type: string
      requestBody:
        content:
          application/json:
            schema:
              type: object
              properties:
                order.invoice_number:
                  type: string
                  description: >-
                    Same as the request that merchant sent on the payment
                    initiation
                order.amount:
                  type: number
                  description: >-
                    Same as the request that merchant sent on the payment
                    initiation
                customer.id:
                  type: string
                  description: >-
                    Same as the request that merchant sent on the payment
                    initiation
                customer.name:
                  type: string
                  description: >-
                    Same as the request that merchant sent on the payment
                    initiation
                customer.email:
                  type: string
                  description: >-
                    Same as the request that merchant sent on the payment
                    initiation
                transaction.type:
                  type: string
                  description: >-
                    Transaction Type Possible value: SALE, INSTALLMENT,
                    AUTHORIZE, MOTO
                transaction.status:
                  type: string
                  description: >-
                    The transaction status to be use to update the status on
                    merchant side Possible value: SUCCESS, FAILED
                transaction.date:
                  type: string
                  description: >-
                    The date when the transaction is completed by the customer
                    in ISO 8601 format (UTC + 0)
                transaction.original_request_id:
                  type: string
                  description: The request ID that sent when initiate the payment
                service.id:
                  type: string
                  description: The service that is used for the transaction
                acquirer.id:
                  type: string
                  description: The acquirer that processed the transaction
                channel.id:
                  type: string
                  description: The channel of the transaction
                authorize_id:
                  type: string
                  description: Used to capture transaction in Authorize Capture transaction
                card_payment.masked_card_number:
                  type: string
                  description: The masked card number that charged for this transaction
                card_payment.approval_code:
                  type: string
                  description: Acquirer approval code for this transaction
                card_payment.response_code:
                  type: string
                  description: DOKU / Acquirer response code
                card_payment.response_message:
                  type: string
                  description: DOKU / Acquirer response message
                card_payment.issuer:
                  type: string
                  description: Card Issuer
                currency:
                  type: string
                  description: Value will be MYR for Malaysian merchant
                card_payment.brand:
                  type: string
                  description: Brand Card Customer
                card_payment.token_id:
                  type: string
                  description: Token when customer save card
                card_payment.authentication_id:
                  type: string
                  description: Authentication ID for 3DS transaction
                card_payment.three_d_secure_status:
                  type: string
                  description: 3DS Status transaction
                card_payment.identifier:
                  type: string
                  description: Identifier config transaction
                card.token:
                  type: string
                  description: Token when customer save card
                verification.status:
                  type: string
                  description: Verification Status from DOKU's Risk Engine
                verification.reason:
                  type: string
                  description: Reason Verification from DOKU's Risk Engine

                - customer.id

                - card_payment.response_code
                - card_payment.response_message
                - card_payment.issuer
                - card_payment.identifier
                - card_payment.brand
                - card_payment.token_id
                - card_payment.authentication_id
                - card_payment.three_d_secure_status
                - card.token
                - verification.status
                - verification.reason
              required:
                - order.invoice_number
                - order.amount
                - customer.id
                - transaction.original_request_id
                - transaction.date
                - transaction.status
                - transaction.type
                - channel.id
                - acquirer.id
                - service.id
                - currency
                - card_payment.brand
                - card_payment.three_d_secure_status
            examples:
```
